local function posOf(inst: Instance): Vector3?
    if inst:IsA("BasePart") then return inst.Position end
    if inst:IsA("Model") then
        local ok, cf = pcall(inst.GetPivot, inst)
        if ok then return cf.Position end
    end
    return nil
end

-- Egg Detection Core
local function isEgg(inst: Instance): boolean
    if not (inst:IsA("BasePart") or inst:IsA("Model")) then return false end
    if not inst.Name:lower():find("egg") and not inst.Name:lower():find("ovo") then return false end
    local ch = LocalPlayer.Character
    if ch and inst:IsDescendantOf(ch) then return false end
    return true
end

    if isEgg(inst) then eggs[inst] = true end
end

-- Pet Reading & Egg Ranking
local KEYS = { "pet", "reward", "content", "hatch", "result", "item", "unit", "brainrot" }
local function hasKey(name: string): boolean
    local n = name:lower()
    for _, k in ipairs(KEYS) do
        if n:find(k) then return true end
    end
    return false
end

local function readFrom(inst: Instance): string?
    for k, v in pairs(inst:GetAttributes()) do
        if type(v) == "string" and v ~= "" and hasKey(k) then return v end
    end
    for _, d in ipairs(inst:GetDescendants()) do
        if d:IsA("StringValue") and d.Value ~= "" and hasKey(d.Name) then return d.Value end
        if d:IsA("ObjectValue") and d.Value and hasKey(d.Name) then return d.Value.Name end
        if (d:IsA("Model") or d:IsA("BasePart")) and d.Name:lower():find("pet") then return d.Name end
    end
    return nil
end

local function readPet(inst: Instance): string?
    local found = readFrom(inst)
    if found then return found end
    local par = inst.Parent
    if par and par ~= Workspace and par:IsA("Model") then
        for k, v in pairs(par:GetAttributes()) do
            if type(v) == "string" and v ~= "" and hasKey(k) then return v end
        end
    end
    return nil
end

local RANKS = {
    { "secret", 8 }, { "divine", 8 }, { "mythic", 7 }, { "legend", 6 },
    { "epic", 5 }, { "rare", 4 }, { "uncommon", 3 }, { "common", 2 },
}
local NUM_KEYS = { "rarity", "tier", "value", "price", "cost", "worth", "multiplier", "level", "luck" }

local function rankOf(text: string): number
    local t = text:lower()
    for _, r in ipairs(RANKS) do
        if t:find(r[1]) then return r[2] end
    end
    return 0
end

local function scoreEgg(inst: Instance): number
    local rankBest, numBest = rankOf(inst.Name), 0
    local pet = readPet(inst)
    if pet then rankBest = math.max(rankBest, rankOf(pet)) end
    return rankBest * 1e6 + math.min(numBest, 999999)
end

local function bestEgg(from: Vector3): Instance?
    local best, bestScore, bestDist = nil, -1, math.huge
    for inst in pairs(eggs) do
        if inst.Parent then
            local p = posOf(inst)
            if p then
                local s = scoreEgg(inst)
                local d = (p - from).Magnitude
                if s > bestScore or (s == bestScore and d < bestDist) then
                    best, bestScore, bestDist = inst, s, d
                end
            end
        end
    end
    return best
end

local function FastGuideTo(targetCFrame)
    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    for _, part in pairs(char:GetChildren()) do
        if part:IsA("BasePart") then part.CanCollide = false end
    end

    hrp.AssemblyLinearVelocity = Vector3.zero
    hrp.AssemblyAngularVelocity = Vector3.zero
    hrp.CFrame = targetCFrame + Vector3.new(0, 4, 0)

    task.wait(0.05)

    for _, part in pairs(char:GetChildren()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then part.CanCollide = true end
    end
end

local function glide(getTarget: () -> Vector3?, myToken: number): boolean
    local elapsed, sinceRoll = 0, 1
    local cruise = rng:NextNumber(GLIDE_MIN, GLIDE_MAX)
    while alive and token == myToken do
        local dt = RunService.Heartbeat:Wait()
        local root, hum = getRoot(), getHum()
        if not root or not hum or hum.Health <= 0 then return false end
        local target = getTarget()
        if not target then return false end
        local delta = target - root.Position
        local dist = delta.Magnitude
        if dist < 2 then return true end

        elapsed += dt
        sinceRoll += dt
        if sinceRoll >= 0.4 then
            sinceRoll = 0
            cruise = rng:NextNumber(GLIDE_MIN, GLIDE_MAX)
        end

        local ramp = math.clamp(elapsed / 0.3, 0.3, 1)
        local brake = math.clamp(dist / BRAKE_DIST, 0.2, 1)
        local spd = cruise * ramp * brake
        local step = math.min(dist, spd * dt)

        root.CFrame = root.CFrame + delta.Unit * step
        root.AssemblyLinearVelocity = Vector3.zero
    end
    return false
end

local function safeSpot(): Vector3?
    local s = env.MilkEggSaved
    if typeof(s) == "Vector3" then return s end
    return nil
end

task.spawn(function()
    local RegisteredObjects = {}
    local descendants = Workspace:GetDescendants()
    
    for i = 1, #descendants do
        local obj = descendants[i]
        pcall(function()
            if obj and (obj:IsA("Model") or obj:IsA("BasePart")) then
                local nameLower = string.lower(obj.Name)
                if string.find(nameLower, "ovo") or string.find(nameLower, "egg") then
                    if not obj:IsDescendantOf(LocalPlayer.Character) and not RegisteredObjects[obj] then
                        RegisteredObjects[obj] = true

                        local incomeVal = ""
                        for _, desc in pairs(obj:GetDescendants()) do
                            if desc:IsA("TextLabel") and desc.Text ~= "" then
                                local txt = desc.Text
                                if string.find(txt, "$") or string.find(string.lower(txt), "/s") or string.find(string.lower(txt), "sec") then
                                    incomeVal = txt
                                    break
                                end
                            end
                        end

                        RenderEggCardSafe(obj, obj.Name, incomeVal)
                    end
                end
            end
        end)

        if i % 300 == 0 then task.wait() end
    end
end)

CreateActionButton(TeleguiadorTab, "teleguiado ovo", function()
    local root = getRoot()
    if not root then return end
    local egg = bestEgg(root.Position)
    if egg then
        token += 1
        local my = token
        task.spawn(function()
            glide(function()
                if not egg.Parent then return nil end
                local p = posOf(egg)
                return p and (p + Vector3.new(0, 3, 0)) or nil
            end, my)
            if token == my then
                TriggerAutoSwingUI()
            end
        end)
    end
end)

CreateActionButton(TeleguiadorTab, "teleguiado voltar", function()
    local home = safeSpot()
    if not home then return end
    token += 1
    local my = token
    task.spawn(function()
        glide(function() return home end, my)
    end)
end)

CreateActionButton(ConfigTab, "⚡ Ativar FPS Boost Ultra", function()
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 9e9
    settings().Rendering.QualityLevel = 1
    for _, item in pairs(Workspace:GetDescendants()) do
        if item:IsA("BasePart") then
            item.Material = Enum.Material.Plastic
            item.Reflectance = 0
        elseif item:IsA("Decal") or item:IsA("Texture") then
            item:Destroy()
        elseif item:IsA("ParticleEmitter") or item:IsA("Trail") then
            item.Enabled = false
        end
    end
end)

CreateToggleButton(ConfigTab, "🛡️ Anti-Lag (Leveza Total)", false, function(enabled)
    States.AntiLag = enabled
    task.spawn(function()
        while States.AntiLag do
            pcall(function()
                for _, v in pairs(Workspace:GetDescendants()) do
                    if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") then
                        v.Enabled = false
                    end
                end
            end)
            task.wait(1)
        end
    end)
end)

CreateToggleButton(ConfigTab, "⚡ Anti-Crash (Estabilidade)", false, function(enabled)
    States.AntiCrash = enabled
    if enabled then
        pcall(function()
            settings().Rendering.EagerBulkExecution = true
            game:GetService("ScriptContext").Error:Connect(function() end)
        end)
    end
end)

CreateActionButton(ConfigTab, "🧹 Remover Todas as Texturas", function()
    pcall(function()
        for _, obj in pairs(Workspace:GetDescendants()) do
            if obj:IsA("Texture") or obj:IsA("Decal") then
                obj:Destroy()
            elseif obj:IsA("BasePart") then
                obj.Material = Enum.Material.SmoothPlastic
                obj.Reflectance = 0
            end
        end
    end)
end)

    if not gameProcessed and (input.KeyCode == Enum.KeyCode.RightControl or input.KeyCode == Enum.KeyCode.Insert) then
        MainFrame.Visible = not MainFrame.Visible
        OpenButton.Visible = not MainFrame.Visible
    end
end)

local function cleanup()
    alive = false
    token += 1
    for _, c in ipairs(connections) do
        pcall(function() c:Disconnect() end)
    end
    connections = {}
    pcall(function() ScreenGui:Destroy() end)
end
env.MilkEggCleanup = cleanup
